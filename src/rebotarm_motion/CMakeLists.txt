# 文件职责：同时安装运动层 Python 节点和 MoveIt C++ 适配插件。
# 对外接口：保留既有可执行名；导出响应适配器及绑定校验控制器管理插件。
# 安全约束：构建不访问设备；C++ 仅转换协议，数值重定时仍由 Python 算法负责。
# 文件结构：依赖 → Python 安装 → 插件 → 可执行入口及资源。
cmake_minimum_required(VERSION 3.16)
project(rebotarm_motion)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(rebotarm_msgs REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_library(rebotarm_execution_adapters SHARED cpp/execution_adapters.cpp)
target_compile_features(rebotarm_execution_adapters PUBLIC cxx_std_17)
ament_target_dependencies(rebotarm_execution_adapters rclcpp pluginlib moveit_core moveit_msgs rebotarm_msgs)
pluginlib_export_plugin_description_file(moveit_core execution_adapters.xml)
install(TARGETS rebotarm_execution_adapters LIBRARY DESTINATION lib)
function(python_entry name module)
  set(entry "${CMAKE_CURRENT_BINARY_DIR}/${name}")
  file(WRITE "${entry}" "#!${Python3_EXECUTABLE}\nfrom ${module} import main\nif __name__ == '__main__':\n    main()\n")
  # 符号链接安装指向构建目录，生成的脚本本身也必须可执行。
  file(CHMOD "${entry}" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(PROGRAMS "${entry}" DESTINATION lib/${PROJECT_NAME})
endfunction()
python_entry(PoseCompatNode rebotarm_motion.pose_compat_node)
python_entry(TrajectoryPreparationNode rebotarm_motion.trajectory_preparation_node)
python_entry(TrajectoryRetimingNode rebotarm_motion.trajectory_retiming_node)
python_entry(PoseExecutionNode rebotarm_motion.pose_execution_node)
python_entry(rebotarm_mujoco_moveit_acceptance rebotarm_motion.mujoco_moveit_acceptance)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
